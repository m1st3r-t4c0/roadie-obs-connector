cmake_minimum_required(VERSION 3.28)
project(roadie-signal VERSION 0.1.9 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)

# Use official packages, or the explicitly prepared local host SDK.
set(ROADIE_OBS_SOURCE_DIR "" CACHE PATH "Matching OBS source tree for local host SDK")
if(ROADIE_OBS_SOURCE_DIR)
  include(cmake/HostSdk.cmake)
else()
  find_package(libobs 30 REQUIRED)
  find_package(obs-frontend-api REQUIRED)
endif()
find_package(Qt6 6.5.3 REQUIRED COMPONENTS Widgets)
set(OBS_BROWSER_PANEL_INCLUDE_DIR "" CACHE PATH "Directory containing the matching browser-panel.hpp")
if(NOT EXISTS "${OBS_BROWSER_PANEL_INCLUDE_DIR}/browser-panel.hpp")
  message(FATAL_ERROR "Supply OBS_BROWSER_PANEL_INCLUDE_DIR from the matching obs-browser source/SDK; no substitute ABI declarations are used.")
endif()
set(ROADIE_SIGNAL_ORIGIN "https://dashboard.roadie.gg" CACHE STRING "Trusted Signal origin; use a separate build for local development")
if(NOT ROADIE_SIGNAL_ORIGIN MATCHES "^https://[A-Za-z0-9.-]+(:[0-9]+)?$" AND
   NOT ROADIE_SIGNAL_ORIGIN MATCHES "^http://(localhost|127\\.0\\.0\\.1)(:[0-9]+)?$")
  message(FATAL_ERROR "Signal origin must be an HTTPS origin or a loopback HTTP development origin, without a path.")
endif()

add_library(roadie-signal MODULE src/plugin.cpp src/scene-events.cpp src/scene-events.hpp src/signal-dock.cpp src/signal-dock.hpp src/audio-tap.cpp src/audio-tap.hpp src/stream-health.cpp src/stream-health.hpp src/crash-reports.cpp src/crash-reports.hpp)
target_include_directories(roadie-signal PRIVATE "${OBS_BROWSER_PANEL_INCLUDE_DIR}")
target_link_libraries(roadie-signal PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Widgets)
target_compile_definitions(roadie-signal PRIVATE ROADIE_SIGNAL_ORIGIN="${ROADIE_SIGNAL_ORIGIN}" ROADIE_SIGNAL_VERSION="${PROJECT_VERSION}")
if(MSVC)
  target_compile_options(roadie-signal PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(roadie-signal PRIVATE -Wall -Wextra -Wpedantic)
endif()
set_target_properties(roadie-signal PROPERTIES PREFIX "")
if(WIN32)
  install(TARGETS roadie-signal RUNTIME DESTINATION obs-plugins/64bit LIBRARY DESTINATION obs-plugins/64bit)
else()
  message(FATAL_ERROR "This first build layout targets Windows x64. Other platforms need a verified packaging path.")
endif()
